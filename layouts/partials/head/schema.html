{{- /* JSON-LD structured data: BlogPosting for article pages, WebSite for the home page. Kept deliberately small — every field here is one the theme can actually verify from configuration, so it does not advertise things the site does not have. */ -}} {{- $brand := site.Params.brand | default site.Title -}} {{- $author := site.Params.author.name | default $brand -}} {{- if .IsPage -}} {{- $author = .Params.author | default $author -}} {{- $desc := .Description | default (.Summary | plainify) | chomp | truncate 200 "" -}} {{- $schema := dict "@context" "https://schema.org" "@type" "BlogPosting" "headline" (.Title | plainify | truncate 110 "") "description" $desc "datePublished" (.Date.Format "2006-01-02T15:04:05Z07:00") "dateModified" (.Lastmod.Format "2006-01-02T15:04:05Z07:00") "inLanguage" site.Language.Locale "mainEntityOfPage" (dict "@type" "WebPage" "@id" .Permalink) "author" (dict "@type" "Person" "name" $author "url" site.BaseURL) "publisher" (dict "@type" "Organization" "name" $brand) -}} {{- $img := .Params.image | default site.Params.og_image -}} {{- with $img }}{{ $schema = merge $schema (dict "image" (. | absURL)) }}{{ end -}} {{- else if .IsHome -}} {{- $schema := dict "@context" "https://schema.org" "@type" "WebSite" "name" $brand "description" (site.Params.description | default "" | plainify | chomp) "url" site.BaseURL "inLanguage" site.Language.Locale -}} {{- end -}}
